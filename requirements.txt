# 说明：
# - 核心依赖跨平台；Windows 专属依赖带 sys_platform 标记，

# GUI 框架
PySide6~=6.5.0                # Qt6 图形界面库

# HTTP 网络库
requests~=2.31.0              # HTTP 请求库

# 图像处理
Pillow~=10.0.0                # 图像处理库
numpy~=1.24.0                 # 数值计算库

# 二维码识别
pyzbar~=0.1.9                 # 二维码解码库（需要系统 libzbar）
opencv-contrib-python~=4.8.0  # OpenCV + WeChat QR 识别器（推荐）
# 或者使用标准版（不包含 WeChat QR）：
# opencv-python~=4.8.0

# Windows 专用（截图/系统 API）
pywin32~=306; sys_platform == 'win32'   # Windows API
dxcam~=0.0.5; sys_platform == 'win32'   # DXGI GPU 加速截图（推荐）
mss~=9.0.1; sys_platform == 'win32'     # 快速截图库（备用）

# ==========================================
# 可选依赖
# ==========================================

# HTTP/2 支持（kuro_api 启动时自动探测，未安装则回退 HTTP/1.1）
# requests-http2>=0.1.0

# GeeTest 验证码窗口需要 QtWebEngine（通常随 PySide6 一起提供，
# 部分精简环境缺系统库时验证码功能降级，见 ui/geetest_dialog.py）
# PySide6-QtWebEngine~=6.5.0; sys_platform == 'win32'
